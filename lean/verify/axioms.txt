def LeanAndrews.ClassicalIdentities : Prop :=
∀ (d n : ℕ), d = 2 ∨ d = 3 → 1 ≤ n → Andrews.Small.qCount d n = Andrews.Small.QCount d n
def LeanAndrews.Pentagonal.EulerRecurrence : Prop :=
∀ (n : ℕ),
  0 < n → ↑(LeanAndrews.Pentagonal.ordinary n) = LeanAndrews.Pentagonal.eulerSum LeanAndrews.Pentagonal.ordinary n 0 n
LeanAndrews.andrews : LeanAndrews.Pentagonal.EulerRecurrence →
  LeanAndrews.ClassicalIdentities →
    ∀ {d n : ℕ},
      2 ≤ d →
        1 ≤ n →
          Andrews.Small.QCount d n ≤ Andrews.Small.qCount d n ∧
            (Andrews.Small.qCount d n = Andrews.Small.QCount d n ↔ d = 2 ∨ d = 3 ∨ n ≤ d + 3 ∨ d = 4 ∧ n = 12)
LeanAndrews.strengthened : LeanAndrews.Pentagonal.EulerRecurrence →
  ∀ {d n : ℕ},
    4 ≤ d →
      1 ≤ n →
        (n ≠ d + 2 → Andrews.Small.QCount d n ≤ Andrews.Small.hCount (d + 1) n) ∧
          (n = d + 2 → Andrews.Small.hCount (d + 1) n = 1 ∧ Andrews.Small.qCount d n = 2 ∧ Andrews.Small.QCount d n = 2)
